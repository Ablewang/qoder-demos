function freshFeedback(g: Game): Feedback | null {
  return feedback && g.elapsed - feedback.at < REACT_SECONDS ? feedback : null;
}
function seatIndex(who: Relative): number {
  return who === 'table' ? -1 : seats.indexOf(who);
}
function seatPerson(who: Relative): void {
  const g = game!;
  if (who === 'table' || seats.includes(who)) return;
  const busy = new Set(g.active.map(q => q.asker));
  const weight = (i: number) => (busy.has(seats[i]) ? 100 : 0) + (ghost?.seat === i ? 10 : 0) + (g.allies.some(a => !a.used && a.asker === seats[i]) ? 3 : 0) + (g.angry.has(seats[i]) ? 3 : 0) + (seats[i] === 'nainai' ? 1 : 0);
  const index = weight(1) <= weight(0) ? 1 : 0;
  if (weight(index) >= 100) return;
  if (ghost?.seat === index) ghost = null;
  seats[index] = who;
}
function syncSeats(): void {
  const g = game!;
  for (const q of g.active) seatPerson(q.asker);
  if (ghost && (g.elapsed >= ghost.until || g.phase !== 'playing' || g.active.length >= 2 || g.active.some(q => seatIndex(q.asker) === ghost!.seat))) ghost = null;
}

function captureFeedback(): void {
  const g = game!;
  const events = g.log.slice(seenEvents);
  seenEvents = g.log.length;
  const relevant = events.filter(e => e.uid !== undefined && e.outcome !== 'unanswered');
  const answer = relevant.filter(e => e.type === 'answer').slice(-1)[0];
  const resolved = relevant.filter(e => e.type === 'resolved').slice(-1)[0];
  const effect = relevant.find(e => e.type === 'flop' || e.type === 'critical');
  const event = resolved?.outcome === 'ally' ? resolved : effect ?? resolved ?? answer;
  if (!event?.qid || event.uid === undefined || !event.asker) return;
  const question = questions.find(q => q.id === event.qid)!;
  const who = event.asker;
  const name = nameOf(who);
  const card = answer?.uid === event.uid ? answer.card : event.card;
  const outcome = resolved?.uid === event.uid ? resolved.outcome : undefined;
  const remaining = g.active.find(q => q.uid === event.uid)?.hp;
  const warm = question.kind === 'warm';
  const boss = question.kind === 'boss';
  const kind: Kind = event.type === 'flop' || event.type === 'critical' ? event.type : outcome === 'ally' ? 'ally' : outcome === 'missed' || outcome === 'warm-missed' ? 'missed' : card ?? 'taiji';
  const angry = g.angry.has(who);
  let label: string;
  let summary: string;
  if (kind === 'flop') { label = '发疯翻车'; summary = '心态 -10 · 引来追问，正在排队'; }
  else if (kind === 'critical') { label = '发疯暴击'; summary = outcome ? '挡下了 · 全桌安静 3 秒' : `全桌安静 3 秒 · 还剩 ${remaining} 血`; }
  else if (kind === 'ally') { label = `${name}帮腔`; summary = '替你挡下这一问'; }
  else if (kind === 'missed') { label = warm ? '关心先收下' : '这句没接住'; summary = warm ? '暖心问不扣心态' : `心态 -${stats[question.kind].hit}`; }
  else {
    const verb = { taiji: '太极', counter: '反问', crazy: '发疯', sincere: '真心' }[kind];
    if (outcome === 'calmed') { label = '气氛缓和'; summary = '化解 · 怒气消了'; }
    else if (outcome === 'sincere') {
      label = boss ? '真心感化' : '真心接住';
      summary = boss ? '全桌感化 · 碰个杯' : warm ? '心态 +20 · 真心牌返还 1 张' : g.allies.some(a => !a.used && a.asker === who) ? `化解 · ${name}站你这边了` : '化解 · 真心不等于让步';
    } else if (outcome === 'blocked') {
      label = `${verb}挡下`;
      summary = kind === 'counter' && angry ? `挡下 · ${name}下一问 +1 血` : '挡下 · 边界说清了';
    } else {
      label = kind === 'crazy' ? '发疯逗笑' : kind === 'sincere' ? '真心感化' : `${verb}接住`;
      summary = kind === 'sincere' && boss ? `感化 3 点 · 心态 +8 · 还剩 ${remaining} 血` : kind === 'counter' && angry ? `还剩 ${remaining} 血 · ${name}记着呢` : `还剩 ${remaining} 血`;
    }
  }
  if (kind !== 'ally') label = `${label} · ${name}`;
  feedback = { uid: event.uid, sequence: g.log.indexOf(event) + 1, asker: who, label, summary, answer: kind === 'ally' ? event.text : answer?.text ?? '', kind, card, warm, at: g.elapsed };
  if (kind === 'ally') { seatPerson(who); audio.play('sincere'); }
  const seat = seatIndex(who);
  if (outcome && kind !== 'ally' && seat >= 0) {
    const stamp = outcome === 'calmed' ? '缓和' : outcome === 'sincere' ? '化解' : outcome === 'blocked' ? '挡下' : '过去了';
    ghost = { uid: event.uid, seat, asker: who, text: kind === 'missed' ? (warm ? '（这份关心，下回慢慢聊）' : '（这句没接上，先夹口菜）') : feedback.answer, stamp, until: g.elapsed + GHOST_SECONDS };
  }
}

function poseFor(who: Seated, g: Game): SeatPose {
  const target = g.target;
  const fresh = freshFeedback(g);
  if (g.freeze > 0) return { gesture: 'hold', gaze: 'you', face: 'flat' };
  if (fresh && fresh.asker === who) {
    if (fresh.kind === 'ally') return { gesture: 'cheer', gaze: 'you', face: 'happy' };
    if (fresh.kind === 'flop' || (fresh.kind === 'counter' && g.angry.has(who))) return { gesture: 'cross', gaze: 'you', face: 'huff' };
    if (fresh.kind === 'missed') return { gesture: 'hold', gaze: 'down', face: 'flat' };
    if (fresh.kind === 'crazy') return { gesture: 'reply', gaze: 'you', face: 'happy' };
    if (fresh.kind === 'sincere') return { gesture: 'reply', gaze: 'you', face: 'smile' };
    return { gesture: 'reply', gaze: 'you', face: 'flat' };
  }
  if (target?.asker === who) return { gesture: 'ask', gaze: 'you', face: who === 'nainai' ? 'smile' : target.maxHp > stats[target.question.kind].hp ? 'huff' : 'ask', lean: true };
  if (g.active.some(q => q.asker === who)) return { gesture: 'hold', gaze: 'in', face: 'flat' };
  if (g.angry.has(who)) return { gesture: 'cross', gaze: 'you', face: 'huff' };
  if (g.allies.some(a => !a.used && a.asker === who)) return { gesture: 'hold', gaze: 'you', face: 'smile' };
  if (target) return { gesture: 'hold', gaze: 'in', face: 'flat' };
  return { gesture: 'hold', gaze: 'down', face: 'smile' };
}
function noteFor(who: Seated, g: Game): string {
  const fresh = freshFeedback(g);
  if (g.freeze > 0) return '愣住了';
  if (fresh && fresh.asker === who) {
    const notes: Record<Kind, string> = { ally: '替你挡了一问', flop: '要追问', critical: '愣住了', missed: fresh.warm ? '先吃热乎的' : '先吃饭', crazy: '笑出了声', sincere: '点点头', taiji: '没再追', counter: g.angry.has(who) ? '记着这句' : '没再追' };
    return notes[fresh.kind];
  }
  if (g.target?.asker === who) return '正在问你';
  if (g.active.some(q => q.asker === who)) return '等你回话';
  if (g.angry.has(who)) return '憋着下一问';
  if (g.allies.some(a => !a.used && a.asker === who)) return '站你这边';
  return g.target ? `看着${nameOf(g.target.asker)}` : '在夹菜';
}

function cast(g: Game, trio: boolean): string {
  if (trio) {
    const third = (['sangu', 'erjiu', 'gufu', 'ershen'] as Seated[]).find(who => !seats.includes(who))!;
    const crew = [seats[0], third, seats[1]];
    return `<div class="cast trio">${crew.map((who, i) => `<div class="seat" data-person="${who}"><div class="person lean">${svg(person(who, { gesture: 'ask', gaze: 'you', face: g.freeze > 0 ? 'flat' : who === 'nainai' ? 'smile' : 'ask' }), i === 2)}</div></div>`).join('')}<small class="seat-note trio-note">全桌 · ${g.freeze > 0 ? '愣住了' : '一起看着你'}</small></div>`;
  }
  return `<div class="cast">${seats.map((who, i) => {
    const pose = poseFor(who, g);
    return `<div class="seat" data-seat="${i}" data-person="${who}" data-gesture="${pose.gesture}" data-gaze="${pose.gaze}"><div class="person${pose.lean ? ' lean' : ''}">${svg(person(who, pose), i === 1)}</div><small class="seat-note">${nameOf(who)} · ${noteFor(who, g)}</small></div>`;
  }).join('')}</div>`;
}

function kindTag(q: ActiveQuestion): string {
  if (q.maxHp > stats[q.question.kind].hp) return '+1 血';
  return { normal: '', fast: '急问', tough: '棘手', warm: '暖心', boss: '压轴', followup: '追问' }[q.question.kind];
}

  const name = nameOf(q.asker);
  const tag = kindTag(q);
  return `<button class="question ${q.question.kind}" data-target="${q.uid}" aria-pressed="${current}" aria-label="${name}问：${esc(q.question.text)}，${q.hp}/${q.maxHp} 血${current ? '，当前回答对象' : '，点选改答这一句'}">${speechOutline(tail)}<span class="question-name"><span class="asker-name"><span class="asker-label">${name}</span>${tag ? `<em>${tag}</em>` : ''}</span><span class="target-mark">${current ? '当前' : '点选'}</span></span><strong class="question-text">${esc(q.question.text)}</strong><span class="question-meta"><span class="pips" aria-hidden="true">${'<i class="on"></i>'.repeat(q.hp)}${'<i></i>'.repeat(q.maxHp - q.hp)}</span><span data-seconds="${q.uid}">耐心 ${Math.ceil(q.remaining)} 秒</span></span><span class="patience" aria-hidden="true"><i data-timer="${q.uid}" style="width:${q.remaining / q.total * 100}%"></i></span></button>`;
}

  return `<div class="reply-ghost" aria-hidden="true">${speechOutline(item.seat === 0 ? 118 : 32, true)}<span class="question-name"><span>你接${nameOf(item.asker)}</span><b class="stamp">${item.stamp}</b></span><span class="ghost-text">${esc(item.text)}</span></div>`;

function talks(g: Game, trio: boolean): string {
  const current = g.targetUid;
  if (g.active.length === 2 && g.active[0].asker === g.active[1].asker) {
    const asker = g.active[0].asker;
    const seat = seatIndex(asker);
    return `<div class="talks same-asker">${speechOutline(seat === 1 ? 104 : 46)}<span class="shared-asker">${nameOf(asker)}连着问了两句 · 点选一句回答</span>${g.active.map(q => bubble(q, 75, q.uid === current)).join('')}</div>`;
  }
  if (trio) return `<div class="talks center">${g.active.map(q => bubble(q, 75, q.uid === current)).join('')}</div>`;
  const shownGhost = ghost && g.elapsed < ghost.until ? ghost : null;
  if (!shownGhost && g.active.length === 1) {
    const q = g.active[0];
    const seat = Math.max(0, seatIndex(q.asker));
    return `<div class="talks single seat-${seat}">${bubble(q, seat === 1 ? 104 : 46, q.uid === current)}</div>`;
  }
  const columns: string[] = ['', ''];
  if (shownGhost) columns[shownGhost.seat] = ghostMarkup(shownGhost);
  for (const q of [...g.active].sort((a, b) => a.uid - b.uid)) {
    const seat = seatIndex(q.asker);
    const column = seat >= 0 && !columns[seat] ? seat : columns[0] ? 1 : 0;
    const tail = seat === column || seat < 0 ? 75 : column === 0 ? 128 : 22;
    columns[column] = bubble(q, tail, q.uid === current);
  }
  return `<div class="talks">${columns.map(content => `<div class="talk-column">${content}</div>`).join('')}</div>`;
}

function statusText(g: Game): string {
  const t = g.target;
  const fresh = freshFeedback(g);
  if (g.phase === 'reward') return '选一份补给，再上下一道菜';
  if (g.reading > 0) return fresh && t ? `接${nameOf(t.asker)}这句 · 免费读 ${Math.ceil(g.reading)} 秒` : `免费读牌 ${Math.ceil(g.reading)} 秒 · 倒计时暂停`;
  if (g.freeze > 0) return `全桌安静 ${Math.ceil(g.freeze)} 秒 · 倒计时暂停`;
  if (fresh) return fresh.summary;
  if (!t) return '趁空夹口菜';
  if (g.cooldown > 0) return `太极冷却 ${g.cooldown.toFixed(1)} 秒 · 其他牌可出`;
  return '读完就出一张牌';
}

'current' : 'previous') : 'none'}"><div class="reading-copy" aria-live="polite"><span class="reply-to">${esc(caption)}</span><span data-live="status">${esc(statusText(g))}</span></div>${g.reading > 0 && g.phase === 'playing' ? '<button data-action="read">读好了 ↗</button>' : ''}</div>`;

function effectLine(card: Card, g: Game): string {
  const t = g.target;
  if (card !== 'taiji' && g.charges[card] <= 0) return card === 'counter' ? '已用完' : '已用完 · 补给可加';
  if (card === 'taiji') return '1 点伤害 · 不限次';
  if (card === 'counter') return t && (t.asker === 'nainai' || t.asker === 'table') ? '3 点伤害 · 不记仇' : '3 点伤害 · 下一问 +1 血';
  if (card === 'crazy') return '随机 · 可能翻车';
  if (!t) return '化解 · 可以结盟';
  if (t.question.kind === 'boss') return '感化 3 点 · 心态 +8';
  if (t.question.kind === 'warm') return '暖心 · 返还 1 张';
  if (g.angry.has(t.asker) || t.maxHp > stats[t.question.kind].hp) return '缓和 · 不结盟';
  return '化解 · 可以结盟';
}
function hand(g: Game): string {
  const t = g.target;
  const fresh = freshFeedback(g);
  return cards.map(card => {
    const state = !t || g.phase !== 'playing' || g.paused ? 'wait' : card !== 'taiji' && g.charges[card] <= 0 ? 'empty' : card === 'taiji' && g.cooldown > 0 ? 'cooldown' : 'ready';
    const charge = card === 'taiji' ? (state === 'cooldown' ? `冷却 <b data-live="cooldown">${g.cooldown.toFixed(1)}</b>` : '<b>∞</b>') : `余 <b>${g.charges[card]}</b>`;
    const text = t ? t.question.answers[card] : cardHints[card];
    const said = !!fresh && fresh.card === card && !!t && fresh.uid === t.uid;
    const effect = effectLine(card, g);
    const parts = effect.split(' · ');
    const effectMarkup = parts.map((part, i) => `<span>${part}${i < parts.length - 1 ? ' ·' : ''}</span>`).join(' ');
    return `<button class="answer-card" data-card="${card}" data-state="${state}"${said ? ' data-said="true"' : ''}${g.canPlay(card) ? '' : ' disabled'} aria-label="${cardNames[card]}牌${said ? '，刚说完' : ''}：${esc(text)}。${effect}">${said ? '<b class="said-stamp" aria-hidden="true">刚说完</b>' : ''}<span class="card-top"><span class="card-icon">${svg(cardIcon(card))}</span><b class="card-name">${cardNames[card]}</b><span class="charges">${charge}</span></span><span class="answer-text">${esc(text)}</span><small class="card-effect">${effectMarkup}</small></button>`;
  }).join('');
}
function footnote(g: Game): string {
  const allies = g.allies.filter(a => !a.used).map(a => nameOf(a.asker));
  const angry = [...g.angry].map(nameOf);
  const parts: string[] = [];
  if (allies.length) parts.push(`${allies.join('、')}站你这边，会替你挡一问`);
  if (angry.length) parts.push(`${angry.join('、')}记着反问，下一问 +1 血`);
  return parts.length ? parts.join('；') : '真心能结盟，坚定拒绝也算答好。';
}
function pauseDialog(): string {

function topAskers(): Seated[] {
  const counts = new Map<Seated, number>();
  for (const e of game!.log) if (e.type === 'resolved' && e.asker && e.asker !== 'table' && e.outcome !== 'unanswered') counts.set(e.asker, (counts.get(e.asker) ?? 0) + 1);
  const ranked = [...counts.entries()].sort((a, b) => b[1] - a[1]).map(entry => entry[0]);
  const pair = [...ranked, ...seats, 'nainai' as Seated, 'dayi' as Seated].filter((who, i, all) => all.indexOf(who) === i).slice(0, 2);
  return pair.includes('nainai') && pair[0] !== 'nainai' ? [pair[1], pair[0]] : pair;
}
function favoriteCard(): Card {
  const counts: Record<Card, number> = { taiji: 0, counter: 0, crazy: 0, sincere: 0 };
  for (const e of game!.log) if (e.type === 'answer' && e.card) counts[e.card]++;
  return cards.reduce((best, card) => counts[card] > counts[best] ? card : best, 'taiji' as Card);
}

function gameSignature(): string {
  const g = game!;
  return JSON.stringify([g.phase, g.wave, g.paused, g.reading > 0, g.freeze > 0, g.cooldown > 0, g.lock <= 0, g.targetUid, g.mood, g.active.map(q => [q.uid, q.hp]), g.charges, g.allies, [...g.angry], g.log.length, seats, ghost?.uid ?? 0, freshFeedback(g)?.sequence ?? 0, audio.muted]);
}
function focusKey(el: Element | null): string {
  if (!(el instanceof HTMLElement) || !app.contains(el)) return '';
  for (const key of ['retry', 'action', 'card', 'target', 'identity', 'reward']) {
    const value = el.dataset[key];
    if (value) return `[data-${key}="${value}"]`;
  }
  return el.tagName === 'SUMMARY' ? 'summary' : '';
}
function modalItems(box: HTMLElement): HTMLElement[] {
  return [...box.querySelectorAll<HTMLElement>('button:not([disabled]), summary, a[href]')].filter(el => el.getClientRects().length > 0);
}
function render(): void {
  const hadModal = !!app.querySelector('.modal');
  const focused = focusKey(document.activeElement);
  app.innerHTML = screen === 'home' ? home() : screen === 'game' ? playing() : screen === 'result' ? result(game!.report!) : posterPage();
  if (notice) {
    const status = document.createElement('p'); status.className = 'notice'; status.role = 'status'; status.textContent = notice; app.append(status);
  }
  updateTimers();
  const box = app.querySelector<HTMLElement>('.modal-box');
  if (box && !hadModal) box.querySelector<HTMLElement>('[data-action]')?.focus({ preventScroll: true });
  else if (!box && hadModal && modalReturn) app.querySelector<HTMLElement>(modalReturn)?.focus({ preventScroll: true });
  else if (focused) app.querySelector<HTMLElement>(box ? `.modal-box ${focused}` : focused)?.focus({ preventScroll: true });
  if (!box) modalReturn = '';
  if (screen !== 'game') return;
  signature = gameSignature();
  const seatKey = `${games}:${seats.join()}`;
  if (renderedSeats && renderedSeats.split(':')[0] === String(games) && seatKey !== renderedSeats && !reduced()) {
    const before = renderedSeats.split(':')[1].split(',');
    seats.forEach((who, i) => { if (before[i] !== who) app.querySelector(`[data-seat="${i}"] .person`)?.animate([{ transform: 'translateY(24px)', opacity: 0 }, { transform: 'translateY(0)', opacity: 1 }], { duration: 260, easing: 'ease-out' }); });
  }
  renderedSeats = seatKey;
  if (feedback && feedback.sequence !== animatedSequence) {
    animatedSequence = feedback.sequence;
    if (!reduced()) {
      app.querySelector('.read-row')?.animate([{ backgroundColor: '#f6d9b6' }, { backgroundColor: '#f1e4cc' }], { duration: 420, easing: 'ease-out' });
      const seat = seatIndex(feedback.asker);
      if (seat >= 0) app.querySelector(`[data-seat="${seat}"] .person`)?.animate([{ transform: 'translateY(-5px)' }, { transform: 'translateY(0)' }], { duration: 220, easing: 'ease-out' });
    }
  }
}
function updateTimers(): void {
  if (screen !== 'game' || !game) return;
  const g = game;
  const status = app.querySelector('[data-live="status"]'); if (status) status.textContent = statusText(g);
  const cooldown = app.querySelector('[data-live="cooldown"]'); if (cooldown) cooldown.textContent = g.cooldown.toFixed(1);
  for (const q of g.active) {
    const bar = app.querySelector<HTMLElement>(`[data-timer="${q.uid}"]`); if (bar) bar.style.width = `${q.remaining / q.total * 100}%`;
    const seconds = app.querySelector(`[data-seconds="${q.uid}"]`); if (seconds) seconds.textContent = `耐心 ${Math.ceil(q.remaining)} 秒`;
  }
}
function start(): void {
  game = new Game(selected, Number.isFinite(seedParam) && seedParam > 0 ? seedParam + games : Date.now());
  games++;
  if ((import.meta as ImportMeta & { env: { DEV: boolean } }).env.DEV && params.has('probe')) (window as unknown as { __rtdGame: Game }).__rtdGame = game;
  screen = 'game'; signature = ''; seenEvents = 0; feedback = null; ghost = null; animatedSequence = 0; quoteIndex = 0; poster = null; notice = ''; posterError = ''; rulesOpen = false;
  seats = ['nainai', 'dayi'];
  renderedSeats = '';
  syncSeats(); render(); window.scrollTo(0, 0); void audio.resume().catch(soundError);
}
function soundError(): void { notice = '浏览器暂未允许音效，仍可静音游玩。'; render(); }
function toResult(): void {
  screen = 'result'; quoteIndex = 0; notice = ''; posterError = ''; audio.play('end'); render(); window.scrollTo(0, 0);
}
function pauseGame(): void {
  if (!game || game.paused || game.phase !== 'playing') return;
  if (!modalReturn) modalReturn = '[data-action="pause"]';
  pauseRulesOpen = false;
  game.togglePause(); audio.suspend(); render();
}
function resumeGame(): void {
  if (game?.paused) game.togglePause();
  void audio.resume().catch(soundError); render();
}
function closeRules(): void { rulesOpen = false; render(); }
function savePoster(): void {
  posterError = ''; notice = '';
  const report = game!.report!;
  try {
    poster = createPoster(report, report.quotes[quoteIndex % report.quotes.length], { cast: topAskers(), card: favoriteCard(), identity: identityName(game!.identity), spoken: game!.log.some(event => event.type === 'answer') });
  } catch {
    poster = null;
    posterError = '战报图这次没生成出来，可能是浏览器内存或权限限制。成绩还在，点“重试生成”再画一次。';
    render();
    app.querySelector<HTMLElement>('[data-retry]')?.focus({ preventScroll: true });
    app.querySelector('.poster-error')?.scrollIntoView({ block: 'nearest' });
    return;
  }
  screen = 'poster'; render(); window.scrollTo(0, 0);
}
app.addEventListener('click', event => {
  const button = (event.target as Element).closest<HTMLElement>('button, a');
  if (!button || button.hasAttribute('disabled')) return;
  const identity = button.dataset.identity;
  if (identity) { selected = identity as Identity; render(); return; }
  if (button.dataset.target) { game?.select(Number(button.dataset.target)); signature = ''; render(); return; }
  if (button.dataset.card) {
    if (game?.play(button.dataset.card as Card)) {
      const effect = game.lastEffect;
      audio.play(effect === 'critical' ? 'critical' : effect === 'flop' ? 'flop' : button.dataset.card === 'sincere' ? 'sincere' : 'card');
      captureFeedback(); syncSeats();
      if (game.phase === 'finished') toResult(); else render();
    }
    return;
  }
  if (button.dataset.reward) { game?.reward(button.dataset.reward as 'soup' | 'sincere' | 'crazy'); audio.play('reward'); syncSeats(); signature = ''; render(); return; }
  switch (button.dataset.action) {
    case 'start': case 'restart': start(); break;
    case 'sound': void audio.toggle().then(() => { notice = ''; render(); }).catch(soundError); break;
    case 'rules': rulesOpen = true; modalReturn = '[data-action="rules"]'; render(); break;
    case 'close-rules': closeRules(); break;
    case 'pause': pauseGame(); break;
    case 'resume': resumeGame(); break;
    case 'read': game?.skipReading(); signature = ''; render(); break;
    case 'finish': game?.finish(); if (game?.phase === 'finished') toResult(); break;
    case 'home': screen = 'home'; game = null; notice = ''; posterError = ''; render(); window.scrollTo(0, 0); break;
    case 'quote': quoteIndex++; render(); break;
    case 'poster': savePoster(); break;
    case 'back': screen = 'result'; posterError = ''; render(); window.scrollTo(0, 0); break;
  }
});
document.addEventListener('keydown', event => {
  const box = app.querySelector<HTMLElement>('.modal-box');
  if (!box) return;
  if (event.key === 'Escape') {
    event.preventDefault();
    if (rulesOpen) closeRules(); else resumeGame();
    return;
  }
  if (event.key !== 'Tab') return;
  const items = modalItems(box);
  if (!items.length) return;
  const first = items[0];
  const last = items[items.length - 1];
  const inside = box.contains(document.activeElement);
  if (!inside || (event.shiftKey && document.activeElement === first)) { event.preventDefault(); (inside ? last : first).focus(); }
  else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
});
document.addEventListener('focusin', event => {
  const box = app.querySelector<HTMLElement>('.modal-box');
  if (box && !box.contains(event.target as Node)) modalItems(box)[0]?.focus({ preventScroll: true });
});
app.addEventListener('toggle', event => {
  if (event.target instanceof HTMLDetailsElement) pauseRulesOpen = event.target.open;
}, true);
document.addEventListener('visibilitychange', () => {
  if (document.hidden) {
    if (screen === 'game') pauseGame();
    audio.suspend();
  }
});
let previous = performance.now();
function frame(now: number): void {
  const dt = (now - previous) / 1000; previous = now;
  if (screen === 'game' && game) {
    game.tick(dt); captureFeedback(); syncSeats();
    if (game.phase === 'finished') toResult();
    else {
      const next = gameSignature();
      if (next !== signature) render(); else updateTimers();
    }
  }
  requestAnimationFrame(frame);
}
render(); requestAnimationFrame(frame);
